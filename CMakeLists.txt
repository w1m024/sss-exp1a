cmake_minimum_required(VERSION 3.16)
project(wirelens VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PCAP REQUIRED IMPORTED_TARGET libpcap)
find_package(Threads REQUIRED)
enable_testing()
option(WIRELENS_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(WIRELENS_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(wirelens_core src/packet.cpp src/capture.cpp)
target_include_directories(wirelens_core PUBLIC src)
target_link_libraries(wirelens_core PUBLIC PkgConfig::PCAP Threads::Threads)
target_compile_options(wirelens_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(wirelens src/server.cpp)
target_include_directories(wirelens PRIVATE vendor)
target_compile_definitions(wirelens PRIVATE WIRELENS_WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/web")
target_link_libraries(wirelens PRIVATE wirelens_core)
add_executable(wirelens-capture src/capture_worker.cpp)
target_link_libraries(wirelens-capture PRIVATE PkgConfig::PCAP Threads::Threads)
target_compile_options(wirelens-capture PRIVATE -Wall -Wextra -Wpedantic)
add_executable(packet_tests tests/packet_tests.cpp src/packet.cpp)
target_include_directories(packet_tests PRIVATE src)
target_link_libraries(packet_tests PRIVATE PkgConfig::PCAP)
add_test(NAME packet_protocols COMMAND packet_tests)
add_executable(ipc_tests tests/ipc_tests.cpp)
target_include_directories(ipc_tests PRIVATE src)
target_link_libraries(ipc_tests PRIVATE Threads::Threads)
add_test(NAME capture_ipc COMMAND ipc_tests)

option(WIRELENS_TCP_OWNER "Build the optional TCP owner probe" OFF)
if(WIRELENS_TCP_OWNER)
    find_program(BPF_CLANG clang)
    if(NOT BPF_CLANG)
        message(FATAL_ERROR "Clang is required for WIRELENS_TCP_OWNER")
    endif()
    pkg_check_modules(BPF REQUIRED IMPORTED_TARGET libbpf)
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -dumpmachine
                    OUTPUT_VARIABLE BPF_MULTIARCH OUTPUT_STRIP_TRAILING_WHITESPACE)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/tcp_owner.bpf.o
        COMMAND ${BPF_CLANG} -Werror -O2 -g -target bpf -I/usr/include/${BPF_MULTIARCH}
                -c ${CMAKE_CURRENT_SOURCE_DIR}/src/tcp_owner.bpf.c
                -o ${CMAKE_CURRENT_BINARY_DIR}/tcp_owner.bpf.o
        DEPENDS src/tcp_owner.bpf.c src/tcp_owner_event.h VERBATIM)
    add_custom_target(tcp_owner_bpf DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/tcp_owner.bpf.o)
    add_executable(wirelens-tcp-owner src/tcp_owner.cpp)
    target_link_libraries(wirelens-tcp-owner PRIVATE PkgConfig::BPF)
    target_compile_options(wirelens-tcp-owner PRIVATE -Wall -Wextra -Wpedantic)
    add_dependencies(wirelens-tcp-owner tcp_owner_bpf)
endif()
